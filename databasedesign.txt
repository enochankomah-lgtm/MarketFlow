/*users table design */

CREATE TABLE Users (
    UserID INT PRIMARY KEY AUTO_INCREMENT,

    RoleID INT NOT NULL,

    FirstName VARCHAR(50) NOT NULL,
    LastName VARCHAR(50) NOT NULL,

    JobTitle VARCHAR(100),

    Email VARCHAR(100) NOT NULL UNIQUE,
    Phone VARCHAR(20),

    PasswordHash VARCHAR(255) NOT NULL,

    IsActive BOOLEAN DEFAULT TRUE,

    CreatedAt DATETIME DEFAULT CURRENT_TIMESTAMP,

    UpdatedAt DATETIME DEFAULT CURRENT_TIMESTAMP
        ON UPDATE CURRENT_TIMESTAMP,

    FOREIGN KEY (RoleID)
        REFERENCES Roles(RoleID)
);

/* Roles table */

CREATE TABLE Roles (
    RoleID INT PRIMARY KEY AUTO_INCREMENT,
    RoleName VARCHAR(50) NOT NULL UNIQUE,
    Description VARCHAR(255)
);

When handling an event, use this code to prevent default behavior and to cancel bubbling. An event is raised on the active element focused. By default the event will bubble up to higher levels in the html tree.
event.preventDefault(); // This will prevent the default behavior of a keypress, such as ctrl+s will often trigger saving on a web page.
event.stopPropagation(); // This will prevent bubbling the event higher in the html tree.
